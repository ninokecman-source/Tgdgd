# Emmett sustav na serveru - crontab korisnika `emmett`.
# Ucitaj ga s:  crontab ~/Tgdgd/server/crontab.txt
#
# Server radi u UTC, a tecajevi i podsjetnici se racunaju po lokalnom danu,
# pa TZ mora biti postavljen - inace podsjetnik ode dan ranije ili kasnije.
TZ=Europe/Zagreb
SHELL=/bin/bash

# Prijave iz Zoha u Excel - svakih pola sata, da prijava brzo bude u tablici.
# BRAVA_CEKANJE=300: ako brava nije slobodna za pet minuta, preskoci - sljedeci
# pokusaj je ionako za 25 minuta, pa se poslovi ne nagomilavaju.
*/30 * * * * BRAVA_CEKANJE=300 $HOME/Tgdgd/server/pokreni.sh zoho_to_excel.py >> $HOME/Tgdgd/log.txt 2>&1

# Bankovni izvodi i Solo ponude - svakih 6 sati, pola sata poslije
30 */6 * * * $HOME/Tgdgd/server/pokreni.sh bank_solo/sync.py >> $HOME/Tgdgd/sync.log 2>&1

# Podsjetnici polaznicima
45 8 * * * $HOME/Tgdgd/server/pokreni.sh send_reminders.py >> $HOME/Tgdgd/log.txt 2>&1

# Tablica centrali, dan nakon tecaja
30 9 * * * $HOME/Tgdgd/server/pokreni.sh posalji_tablicu.py >> $HOME/Tgdgd/log.txt 2>&1

# Potvrdnice NEMAJU svoj cron. Nastaju kad na stranici tecaja pritisnes
# "Posalji potvrdnice" - tko naruci slanje, time ih i naruci. Tako se
# ispravak imena u tablici odmah vidi: za novo ime potvrdnice nema, pa
# nastane nova. Dok su se radile same u 10:00, ostajala je stara s krivim
# imenom i nitko to nije imao gdje primijetiti.

# Provjera logova - svaka 3 dana
0 9 */3 * * $HOME/Tgdgd/server/pokreni.sh provjeri_logove.py >> $HOME/Tgdgd/log.txt 2>&1

# Sinkronizacija i kad nema posla, da Dropbox ne zaostane; uz nju se osvjezi
# i snimka koju cita web pregled. Preskace se ako neki drugi posao radi.
*/20 * * * * flock -n $HOME/.emmett.lock $HOME/Tgdgd/server/osvjezi.sh >/dev/null 2>>$HOME/Tgdgd/log.txt
